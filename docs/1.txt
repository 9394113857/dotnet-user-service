PowerShell command:
# CREATE a new user
# This sends JSON data to the User API
Invoke-RestMethod `
  -Uri "http://localhost:5228/api/users" `
  -Method POST `
  -ContentType "application/json" `
  -Body '{"username":"bro-user","email":"bro@example.com"}'

# READ ALL users
# This retrieves every record from the Users table
Invoke-RestMethod `
  -Uri "http://localhost:5228/api/users" `
  -Method GET

# READ ONE user
# This retrieves only the user whose ID is 1
Invoke-RestMethod `
  -Uri "http://localhost:5228/api/users/1" `
  -Method GET

# UPDATE user ID 1
# Replace the existing username and email
Invoke-RestMethod `
  -Uri "http://localhost:5228/api/users/1" `
  -Method PUT `
  -ContentType "application/json" `
  -Body '{"username":"bro-updated","email":"updated@example.com"}'

# READ user ID 1 again
# This verifies that the UPDATE was persisted in the database
Invoke-RestMethod `
  -Uri "http://localhost:5228/api/users/1" `
  -Method GET

# DELETE user ID 1
# This removes the user from the Users table
Invoke-RestMethod `
  -Uri "http://localhost:5228/api/users/1" `
  -Method DELETE

# READ ALL users one final time
# The deleted user should no longer exist
Invoke-RestMethod `
  -Uri "http://localhost:5228/api/users" `
  -Method GET
